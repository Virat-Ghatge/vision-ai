export interface CenterCoordinates {
  x: number;
  y: number;

export interface ElementMeta {
  tagName: string;
  text: string;
  ariaLabel: string;
  nameAttr: string;
  role: string;
  idAttr: string;
  dataTestId: string;
  href: string;
}

export class ElementRegistry {
  private idToElement = new Map<number, WeakRef<HTMLElement>>();
  private idToMeta = new Map<number, ElementMeta>();
  private elementToId = new WeakMap<HTMLElement, number>();
  private nextId = 1;

  register(element: HTMLElement): number {
    const existingId = this.elementToId.get(element);
    if (existingId !== undefined) {
      this.idToElement.set(existingId, new WeakRef(element));
      this.idToMeta.set(existingId, this.extractMeta(element));
      return existingId;
    }

    const id = this.nextId++;
    this.idToElement.set(id, new WeakRef(element));
    this.idToMeta.set(id, this.extractMeta(element));
    this.elementToId.set(element, id);
    return id;
  }

  private extractMeta(el: HTMLElement): ElementMeta {
    return {
      tagName: el.tagName.toUpperCase(),
      text: (el.textContent || '').replace(/\s+/g, ' ').trim().slice(0, 60),
      ariaLabel: (el.getAttribute('aria-label') || el.getAttribute('title') || '').trim(),
      nameAttr: (el.getAttribute('name') || '').trim(),
      role: (el.getAttribute('role') || '').trim().toLowerCase(),
      idAttr: el.id || '',
      dataTestId: (el.getAttribute('data-testid') || el.getAttribute('data-test') || '').trim(),
      href: (el.getAttribute('href') || '').trim(),
    };
  }

  /**
   * Retrieves an element by ID. If the element was disconnected due to a React/Coursera
   * SPA re-render, automatically self-heals by locating the replacement element in the live DOM.
   */
  getElement(id: number): HTMLElement | null {
    const ref = this.idToElement.get(id);
    if (ref) {
      const el = ref.deref();
      if (el && el.isConnected) {
        return el;
      }
      this.idToElement.delete(id);
    }

    // Self-heal lookup ONLY if the element was NOT explicitly removed cleanly in a unit test
    // Check if we can relocate an exact match by stable attributes on complex SPAs
    const meta = this.idToMeta.get(id);
    if (meta && typeof document !== 'undefined' && document.body) {
      if (meta.idAttr) {
        const byId = document.getElementById(meta.idAttr);
        if (byId) {
          this.idToElement.set(id, new WeakRef(byId));
          this.elementToId.set(byId, id);
          return byId;
        }
      }
      if (meta.dataTestId) {
        const byTestId = document.querySelector<HTMLElement>(
          `[data-testid="${meta.dataTestId}"], [data-test="${meta.dataTestId}"]`
        );
        if (byTestId) {
          this.idToElement.set(id, new WeakRef(byTestId));
          this.elementToId.set(byTestId, id);
          return byTestId;
        }
      }
    }

    return null;
  }

  /**
   * Deep self-healing fallback used by MacroBatchExecutor when a React SPA re-mounts a button/link.
   */
  findReplacementElement(id: number): HTMLElement | null {
    const direct = this.getElement(id);
    if (direct) return direct;

    const meta = this.idToMeta.get(id);
    if (!meta || typeof document === 'undefined' || !document.body) return null;

    const all = collectAllElementsDeep(document.body, 2500);
    for (const candidate of all) {
      if (candidate.tagName.toUpperCase() !== meta.tagName) continue;
      if (meta.ariaLabel) {
        const candAria = (candidate.getAttribute('aria-label') || candidate.getAttribute('title') || '').trim();
        if (candAria === meta.ariaLabel) {
          this.idToElement.set(id, new WeakRef(candidate));
          this.elementToId.set(candidate, id);
          return candidate;
        }
      }
      if (meta.text && meta.text.length >= 3) {
        const candText = (candidate.textContent || '').replace(/\s+/g, ' ').trim().slice(0, 60);
        if (candText === meta.text) {
          this.idToElement.set(id, new WeakRef(candidate));
          this.elementToId.set(candidate, id);
          return candidate;
        }
      }
    }
    return null;
  }

  getId(element: HTMLElement): number | undefined {
    return this.elementToId.get(element);
  }

  getCenterCoordinates(id: number): CenterCoordinates | null {
    const el = this.getElement(id);
    if (!el) return null;

    const rect = el.getBoundingClientRect();
    if (rect.width === 0 && rect.height === 0) {
      return null;
    }

    let offsetX = 0;
    let offsetY = 0;
    try {
      const ownerWin = el.ownerDocument?.defaultView;
      if (ownerWin && ownerWin !== window && ownerWin.frameElement) {
        const frameRect = ownerWin.frameElement.getBoundingClientRect();
        offsetX = frameRect.left;
        offsetY = frameRect.top;
      }
    } catch {}

    return {
      x: Math.round(offsetX + rect.left + rect.width / 2),
      y: Math.round(offsetY + rect.top + rect.height / 2),
    };
  }

  clear(): void {
    this.idToElement.clear();
    this.idToMeta.clear();
    this.nextId = 1;
  }

  size(): number {
    return this.idToElement.size;
  }
}

const INTERACTIVE_TAGS = new Set([
  'A',
  'BUTTON',
  'INPUT',
  'SELECT',
  'TEXTAREA',
  'SUMMARY',
  'DETAILS',
  'LABEL',
  'VIDEO',

export const INTERACTIVE_ROLES = new Set([
  'button',
  'link',
  'checkbox',
  'menuitem',
  'radio',
  'tab',
  'combobox',
  'searchbox',
  'switch',
  'textbox',

const LANDMARK_TAGS = new Set(['H1', 'H2', 'H3']);

const LMS_PRIORITY_REGEX =
  /\b(mark as complete|mark complete|completed|complete|next item|go to next|next|continue|resume|start|launch|submit|play|study material|lecture|reading|video|module|week|lesson|quiz|grade|transcript|syllabus)\b/i;

export function collectAllElementsDeep(
  root: Document | HTMLElement | ShadowRoot,
  maxNodes = 4000
): HTMLElement[] {
  const result: HTMLElement[] = [];
  const startNode = root instanceof Document ? root.body : root;
  if (!startNode) return result;

  const visit = (container: HTMLElement | ShadowRoot) => {
    if (result.length >= maxNodes) return;
    const children = container.children;
    for (let i = 0; i < children.length && result.length < maxNodes; i++) {
      const el = children[i] as HTMLElement;
      if (!el || !el.tagName) continue;
      const tag = el.tagName.toUpperCase();
      if (tag === 'SCRIPT' || tag === 'STYLE' || tag === 'NOSCRIPT' || tag === 'SVG' || tag === 'PATH') {
        continue;
      }

      result.push(el);

      if (el.shadowRoot) {
        visit(el.shadowRoot);
      }

      if (tag === 'IFRAME') {
        try {
          const frameDoc = (el as HTMLIFrameElement).contentDocument;
          if (frameDoc?.body) {
            visit(frameDoc.body);
          }
        } catch {}
      }

      if (el.children && el.children.length > 0) {
        visit(el);
      }
    }
  };

  visit(startNode);
  return result;
}

export function isHighPriorityLMSAction(el: HTMLElement): boolean {
  const tag = el.tagName.toUpperCase();
  if (tag === 'VIDEO' || tag === 'AUDIO') return true;

  const role = (el.getAttribute('role') || '').toLowerCase();
  const isActionable =
    tag === 'BUTTON' ||
    tag === 'A' ||
    tag === 'INPUT' ||
    tag === 'SUMMARY' ||
    INTERACTIVE_ROLES.has(role) ||
    el.hasAttribute('data-testid') ||
    el.hasAttribute('aria-expanded') ||
    el.hasAttribute('aria-checked');

  if (!isActionable) return false;

  const sig = `${el.getAttribute('aria-label') || ''} ${el.getAttribute('title') || ''} ${
    el.getAttribute('data-testid') || ''
  } ${(el.textContent || '').slice(0, 90)}`;

  return LMS_PRIORITY_REGEX.test(sig);
}

export function isElementVisible(
  el: HTMLElement,
  win: Window = window,
  allowBelowFold = false
): boolean {
  if (!el.isConnected) return false;
  if (el.hasAttribute('hidden') || el.closest?.('[hidden]')) return false;
  if (el.getAttribute('aria-hidden') === 'true') return false;

  if (typeof (el as any).checkVisibility === 'function') {
    try {
      if (!(el as any).checkVisibility({ checkOpacity: true, checkVisibilityCSS: true })) {
        return false;
      }
    } catch {}
  } else {
    let cur: HTMLElement | null = el;
    let depth = 0;
    while (cur && depth < 8) {
      if (
        cur.style?.display === 'none' ||
        cur.style?.visibility === 'hidden' ||
        cur.style?.opacity === '0'
      ) {
        return false;
      }
      const curStyle = win.getComputedStyle ? win.getComputedStyle(cur) : null;
      if (curStyle) {
        if (
          curStyle.display === 'none' ||
          curStyle.visibility === 'hidden' ||
          curStyle.opacity === '0'
        ) {
          return false;
        }
      }
      cur = cur.parentElement;
      depth++;
    }
  }

  const rect = el.getBoundingClientRect ? el.getBoundingClientRect() : null;
  if (rect) {
    const vw = win.innerWidth || 1920;
    const vh = win.innerHeight || 1080;

    if (rect.width > 0 || rect.height > 0) {
      const verticalBuffer = allowBelowFold ? vh * 3 : 120;
      if (
        rect.bottom < -80 ||
        rect.top > vh + verticalBuffer ||
        rect.right < -50 ||
        rect.left > vw + 50
      ) {
        return false;
      }
    }
  }

  return true;
}

export function isCandidateNode(el: HTMLElement, win: Window = window): boolean {
  const tagName = el.tagName.toUpperCase();

  if (INTERACTIVE_TAGS.has(tagName)) {
    return true;
  }

  const role = el.getAttribute('role');
  if (role && INTERACTIVE_ROLES.has(role.toLowerCase())) {
    return true;
  }

  if (LANDMARK_TAGS.has(tagName)) {
    return true;
  }

  if (el.getAttribute('contenteditable') === 'true') {
    return true;
  }

  const tabIndex = el.getAttribute('tabindex');
  if (tabIndex !== null && parseInt(tabIndex, 10) >= 0) {
    return true;
  }

  if (
    el.hasAttribute('onclick') ||
    el.hasAttribute('aria-expanded') ||
    el.hasAttribute('aria-checked') ||
    el.hasAttribute('data-action') ||
    el.hasAttribute('data-clickable')
  ) {
    return true;
  }

  if (typeof win !== 'undefined' && win.getComputedStyle) {
    try {
      const curStyle = win.getComputedStyle(el);
      if (curStyle && curStyle.cursor === 'pointer') {
        if (
          el.children.length <= 2 &&
          !el.closest('a, button, [role="button"], [role="link"], [role="menuitem"], [role="tab"]') &&
          (!el.parentElement || win.getComputedStyle(el.parentElement).cursor !== 'pointer')
        ) {
          return true;
        }
      }
    } catch {}
  }

  const className = (el.className || '').toString().toLowerCase();
  if (
    (className.includes('accordion') ||
      className.includes('clickable') ||
      className.includes('btn') ||
      className.includes('weekitem') ||
      className.includes('nav-item')) &&
    el.children.length <= 2 &&
    !el.querySelector('a, button, input, [role="button"], [role="link"]') &&
    !el.closest('a, button, [role="button"], [role="link"]') &&
    (el.textContent || '').trim().length > 0 &&
    (el.textContent || '').trim().length < 90
  ) {
    return true;
  }

  return false;
}

function truncateText(str: string, maxLen = 40): string {
  const cleaned = str.replace(/\s+/g, ' ').trim();
  if (cleaned.length <= maxLen) return cleaned;
  return cleaned.substring(0, maxLen - 3) + '...';
}

export function formatElementDescriptor(el: HTMLElement, id: number): string {
  const tag = el.tagName.toLowerCase();
  const attrs: string[] = [];

  const type = el.getAttribute('type');
  if (type) attrs.push(`type="${type}"`);

  const placeholder = el.getAttribute('placeholder');
  if (placeholder) attrs.push(`placeholder="${truncateText(placeholder, 35)}"`);

  const ariaLabel = el.getAttribute('aria-label') || el.getAttribute('title');
  if (ariaLabel) attrs.push(`label="${truncateText(ariaLabel, 30)}"`);

  const role = el.getAttribute('role');
  if (role && role !== tag) attrs.push(`role="${role}"`);

  const name = el.getAttribute('name');
  if (name) attrs.push(`name="${name}"`);

  const expanded = el.getAttribute('aria-expanded');
  if (expanded === 'true' || expanded === 'false') {
    attrs.push(`expanded=${expanded}`);
  }

  const ariaChecked = el.getAttribute('aria-checked');
  if (ariaChecked === 'true' || ariaChecked === 'false') {
    attrs.push(`checked=${ariaChecked}`);
  }

  if (el.hasAttribute('disabled') || el.getAttribute('aria-disabled') === 'true') {
    attrs.push('disabled');
  }

  if (tag === 'video' || tag === 'audio') {
    const media = el as HTMLMediaElement;
    const ended = media.ended ? 'ended' : media.paused ? 'paused' : 'playing';
    attrs.push(`state="${ended}"`);
  }

  if (
    el instanceof HTMLInputElement ||
    el instanceof HTMLTextAreaElement ||
    el instanceof HTMLSelectElement
  ) {
    if (el.value) {
      attrs.push(`value="${truncateText(el.value, 25)}"`);
    }
    if (el instanceof HTMLInputElement && el.checked) {
      attrs.push('checked');
    }
  }

  const attrStr = attrs.length > 0 ? ` ${attrs.join(' ')}` : '';

  let text = '';
  if (tag === 'select') {
    const sel = el as HTMLSelectElement;
    text =
      sel.selectedIndex >= 0 && sel.options?.[sel.selectedIndex]
        ? truncateText(sel.options[sel.selectedIndex].text || '', 20)
        : '';
  } else if (tag !== 'input') {
    text = truncateText(el.textContent || '', 36);
  }

  if (text) {
    return `[${id}] <${tag}${attrStr}>${text}</${tag}>`;
  }
  return `[${id}] <${tag}${attrStr} />`;
}

export interface DistillOptions {
  maxItems?: number;
  includeHeader?: boolean;
}

export interface DistillResult {
  text: string;
  totalElements: number;
  interactiveCount: number;
}

export function distillViewport(
  root: Document | HTMLElement = document,
  registry: ElementRegistry = globalElementRegistry,
  win: Window = window,
  options?: DistillOptions
): DistillResult {
  const maxItems = options?.maxItems ?? 62;
  const lines: string[] = [];
  let interactiveCount = 0;

  const walker = root instanceof Document ? root.body : root;
  if (!walker) {
    return { text: '(Empty document)', totalElements: 0, interactiveCount: 0 };
  }

  const doc = root instanceof Document ? root : root.ownerDocument;
  const pageTitle = doc?.title?.trim() || '';
  const pageUrl =
    win.location?.href && !win.location.href.startsWith('about:') ? win.location.href : '';

  if (options?.includeHeader ?? true) {
    if (pageTitle || pageUrl) {
      lines.push(`PAGE: ${pageTitle || '(Untitled)'} [${pageUrl || 'current-tab'}]`);
    }
  }

  const elements = collectAllElementsDeep(walker);
  const totalWalked = elements.length;

  const priorityCandidates: HTMLElement[] = [];
  const standardCandidates: HTMLElement[] = [];
  const seenSet = new Set<HTMLElement>();

  for (let i = 0; i < elements.length; i++) {
    const el = elements[i];
    if (seenSet.has(el)) continue;

    if (!isCandidateNode(el, win)) continue;

    const interactiveAncestor = el.parentElement?.closest(
      'button, a, [role="button"], [role="link"], [role="menuitem"], [role="tab"], [role="checkbox"], [role="radio"]'
    );
    if (
      interactiveAncestor &&
      !['INPUT', 'SELECT', 'TEXTAREA', 'VIDEO'].includes(el.tagName.toUpperCase())
    ) {
      continue;
    }

    const isPriority = isHighPriorityLMSAction(el);
    if (!isElementVisible(el, win, isPriority)) continue;

    seenSet.add(el);
    if (isPriority) {
      priorityCandidates.push(el);
    } else {
      standardCandidates.push(el);
    }
  }

  const reservedPriority = new Set(priorityCandidates.slice(0, 25));
  const orderedFinal: HTMLElement[] = [];

  for (let i = 0; i < elements.length; i++) {
    const el = elements[i];
    if (!seenSet.has(el)) continue;
    if (reservedPriority.has(el) || standardCandidates.includes(el)) {
      orderedFinal.push(el);
    }
  }

  for (const pEl of reservedPriority) {
    if (!orderedFinal.includes(pEl)) {
      orderedFinal.push(pEl);
    }
  }

  let currentChars = lines.reduce((acc, l) => acc + l.length + 1, 0);

  for (let i = 0; i < orderedFinal.length; i++) {
    const el = orderedFinal[i];
    const isPriority = reservedPriority.has(el);

    if (
      !isPriority &&
      interactiveCount >= maxItems - reservedPriority.size &&
      interactiveCount >= maxItems
    ) {
      continue;
    }

    if (interactiveCount >= maxItems || Math.ceil(currentChars / 3.8) >= 680) {
      lines.push(`... [${totalWalked - interactiveCount} additional items outside viewport budget]`);
      break;
    }

    const id = registry.register(el);
    const desc = formatElementDescriptor(el, id);
    lines.push(desc);
    currentChars += desc.length + 1;
    interactiveCount++;
  }

  try {
    const textEls = walker.querySelectorAll<HTMLElement>(
      'h1, h2, h3, p, article, [class*="price"], [data-testid*="title"]'
    );
    const snippets: string[] = [];
    let totalChars = 0;
    for (let i = 0; i < textEls.length && totalChars < 350; i++) {
      const tEl = textEls[i];
      if (!isElementVisible(tEl, win)) continue;
      const txt = (tEl.textContent || '').replace(/\s+/g, ' ').trim();
      if (txt.length > 5 && !snippets.some((s) => s.includes(txt) || txt.includes(s))) {
        const piece = txt.slice(0, 100);
        snippets.push(piece);
        totalChars += piece.length;
      }
    }
    if (snippets.length > 0) {
      const estimatedTokens = Math.ceil(currentChars / 3.8);
      if (estimatedTokens < 720) {
        const allowedChars = Math.min(350, Math.floor((770 - estimatedTokens) * 3.8));
        if (allowedChars > 20) {
          lines.push(`PAGE SUMMARY: ${snippets.join(' | ')}`.slice(0, allowedChars));
        }
      }
    }
  } catch {}

  if (interactiveCount === 0 && !lines.some((l) => l.startsWith('PAGE SUMMARY:'))) {
    const textSnippet = (walker.textContent || '').replace(/\s+/g, ' ').trim().slice(0, 300);
    if (textSnippet) {
      lines.push(`(No interactive elements in viewport. Page text summary: "${textSnippet}")`);
    }
  }

  return {
    text: lines.join('\n'),
    totalElements: totalWalked,
    interactiveCount,
  };

export interface ActuatorOptions {

export interface ActionResult {

  method: 'cdp' | 'synthetic';
  error?: string;

const KEY_CODE_MAP: Record<string, { code: string; keyCode: number }> = {
  Enter: { code: 'Enter', keyCode: 13 },
  Tab: { code: 'Tab', keyCode: 9 },
  Escape: { code: 'Escape', keyCode: 27 },
  Backspace: { code: 'Backspace', keyCode: 8 },
  ArrowDown: { code: 'ArrowDown', keyCode: 40 },
  ArrowUp: { code: 'ArrowUp', keyCode: 38 },
};

export class CDPActuator {
  readonly tabId: number;
  private isAttached = false;
  private preferCDP: boolean;

  constructor(options: ActuatorOptions) {
    this.tabId = options.tabId;
    this.preferCDP = options.preferCDP ?? true;

    if (typeof chrome !== 'undefined' && chrome.debugger?.onDetach) {
      chrome.debugger.onDetach.addListener((source, reason) => {
        if (source.tabId === this.tabId) {
          console.warn(`[CDPActuator] Debugger detached from tab ${this.tabId}: ${reason}`);
          this.isAttached = false;
        }
      });
    }
  }

  async attach(): Promise<boolean> {
    if (!this.preferCDP) return false;
    if (this.isAttached) return true;

    if (typeof chrome === 'undefined' || !chrome.debugger) {
      return false;
    }

    try {
      await chrome.debugger.attach({ tabId: this.tabId }, '1.3');
      this.isAttached = true;
      return true;
    } catch (err) {
      console.warn('[CDPActuator] Failed to attach chrome.debugger, falling back to synthetic:', err);
      this.isAttached = false;
      return false;
    }
  }

  async detach(): Promise<void> {
    if (!this.isAttached) return;
    if (typeof chrome === 'undefined' || !chrome.debugger) return;

    try {
      await chrome.debugger.detach({ tabId: this.tabId });
    } catch (err) {
      console.warn('[CDPActuator] Error during detach:', err);
    } finally {
      this.isAttached = false;
    }
  }

  get attached(): boolean {
    return this.isAttached;
  }

  async click(coords?: Coordinates | null, fallbackElement?: HTMLElement | null): Promise<ActionResult> {
    if (coords && this.preferCDP) {
      const attached = await this.attach();
      if (attached) {
        try {
          await this.sendCDPCommand('Input.dispatchMouseEvent', {
            type: 'mousePressed',
            x: coords.x,
            y: coords.y,
            button: 'left',
            clickCount: 1,
          });

          await new Promise((r) => setTimeout(r, 40));

          await this.sendCDPCommand('Input.dispatchMouseEvent', {
            type: 'mouseReleased',
            x: coords.x,
            y: coords.y,
            button: 'left',
            clickCount: 1,
          });

          return { success: true, method: 'cdp' };
        } catch (err) {
          console.warn('[CDPActuator] CDP click failed, falling back to synthetic:', err);
        }
      }
    }

    if (fallbackElement) {
      try {
        try {
          fallbackElement.scrollIntoView({ block: 'center', inline: 'center', behavior: 'instant' });
        } catch {}

        const rect = fallbackElement.getBoundingClientRect
          ? fallbackElement.getBoundingClientRect()
          : { left: 0, top: 0, width: 0, height: 0 };
        const clientX = Math.round((rect.left || 0) + (rect.width || 0) / 2);
        const clientY = Math.round((rect.top || 0) + (rect.height || 0) / 2);

        const win = typeof window !== 'undefined' ? window : undefined;
        const PointerEventClass = typeof PointerEvent !== 'undefined' ? PointerEvent : MouseEvent;

        const baseInit = {
          bubbles: true,
          cancelable: true,
          composed: true,
          clientX,
          clientY,
          button: 0,
          buttons: 1,
          view: win,
        };

        fallbackElement.dispatchEvent(new PointerEventClass('pointerover', { ...baseInit, buttons: 0 }));
        fallbackElement.dispatchEvent(new MouseEvent('mouseover', { ...baseInit, buttons: 0 }));
        fallbackElement.dispatchEvent(new PointerEventClass('pointerdown', baseInit));
        fallbackElement.dispatchEvent(new MouseEvent('mousedown', baseInit));
        if (typeof fallbackElement.focus === 'function') fallbackElement.focus();
        fallbackElement.dispatchEvent(new PointerEventClass('pointerup', { ...baseInit, buttons: 0 }));
        fallbackElement.dispatchEvent(new MouseEvent('mouseup', { ...baseInit, buttons: 0 }));
        // Use a single native .click() so checkboxes and accordions are never double-toggled!
        if (typeof fallbackElement.click === 'function') {
          fallbackElement.click();
        } else {
          fallbackElement.dispatchEvent(new MouseEvent('click', { ...baseInit, buttons: 0 }));
        }

        return { success: true, method: 'synthetic' };
      } catch (err) {
        return {
          success: false,
          method: 'synthetic',
          error: err instanceof Error ? err.message : String(err),
        };
      }
    }

    return { success: false, method: 'synthetic', error: 'Neither coords nor fallbackElement provided' };
  }

  async type(
    text: string,
    coords?: Coordinates | null,
    fallbackElement?: HTMLElement | null
  ): Promise<ActionResult> {
    if (coords && this.preferCDP) {
      const attached = await this.attach();
      if (attached) {
        try {
          await this.click(coords);

          for (const char of text) {
            await this.sendCDPCommand('Input.dispatchKeyEvent', {
              type: 'keyDown',
              text: char,
              unmodifiedText: char,
            });
            await this.sendCDPCommand('Input.dispatchKeyEvent', {
              type: 'keyUp',
            });
          }
          return { success: true, method: 'cdp' };
        } catch (err) {
          console.warn('[CDPActuator] CDP typing failed, falling back to synthetic:', err);
        }
      }
    }

    if (fallbackElement) {
      try {
        fallbackElement.focus();
        if (
          fallbackElement instanceof HTMLInputElement ||
          fallbackElement instanceof HTMLTextAreaElement
        ) {
          fallbackElement.value = text;
          fallbackElement.dispatchEvent(new Event('input', { bubbles: true }));
          fallbackElement.dispatchEvent(new Event('change', { bubbles: true }));
        } else if (fallbackElement.isContentEditable) {
          fallbackElement.textContent = text;
          fallbackElement.dispatchEvent(new Event('input', { bubbles: true }));
        }
        return { success: true, method: 'synthetic' };
      } catch (err) {
        return {
          success: false,
          method: 'synthetic',
          error: err instanceof Error ? err.message : String(err),
        };
      }
    }

    return { success: false, method: 'synthetic', error: 'No target for typing' };
  }

  async pressKey(key: string, fallbackElement?: HTMLElement | null): Promise<ActionResult> {
    const keyInfo = KEY_CODE_MAP[key];

    if (this.preferCDP && (await this.attach())) {
      try {
        await this.sendCDPCommand('Input.dispatchKeyEvent', {
          type: 'rawKeyDown',
          key: keyInfo ? key : key,
          code: keyInfo?.code ?? key,
          windowsVirtualKeyCode: keyInfo?.keyCode ?? 0,
          nativeVirtualKeyCode: keyInfo?.keyCode ?? 0,
        });

        await this.sendCDPCommand('Input.dispatchKeyEvent', {
          type: 'keyUp',
          key: keyInfo ? key : key,
          code: keyInfo?.code ?? key,
          windowsVirtualKeyCode: keyInfo?.keyCode ?? 0,
          nativeVirtualKeyCode: keyInfo?.keyCode ?? 0,
        });

        return { success: true, method: 'cdp' };
      } catch (err) {
        console.warn('[CDPActuator] CDP pressKey failed, falling back to synthetic:', err);
      }
    }

    const target = fallbackElement || (typeof document !== 'undefined' ? document.activeElement : null);
    if (target) {
      target.dispatchEvent(
        new KeyboardEvent('keydown', {
          key,
          code: keyInfo?.code ?? key,
          bubbles: true,
          cancelable: true,
        })
      );
      target.dispatchEvent(
        new KeyboardEvent('keyup', {
          key,
          code: keyInfo?.code ?? key,
          bubbles: true,
          cancelable: true,
        })
      );
      return { success: true, method: 'synthetic' };
    }

    return { success: false, method: 'synthetic', error: 'No active element for key press' };
  }

  async scroll(direction: 'up' | 'down', amount = 400): Promise<ActionResult> {
    const deltaY = direction === 'down' ? amount : -amount;

    if (this.preferCDP && (await this.attach())) {
      try {
        await this.sendCDPCommand('Input.dispatchMouseEvent', {
          type: 'mouseWheel',
          x: 400,
          y: 400,
          deltaX: 0,
          deltaY,
        });
        return { success: true, method: 'cdp' };
      } catch (err) {
        console.warn('[CDPActuator] CDP scroll failed, falling back to synthetic:', err);
      }
    }

    if (typeof window !== 'undefined') {
      window.scrollBy({ top: deltaY, behavior: 'smooth' });
      return { success: true, method: 'synthetic' };
    }

    return { success: false, method: 'synthetic', error: 'Window not available for scroll' };
  }

  private async sendCDPCommand(method: string, params?: Record<string, unknown>): Promise<unknown> {
    if (typeof chrome === 'undefined' || !chrome.debugger) {
      throw new Error('chrome.debugger API is not available');
    }

    return new Promise((resolve, reject) => {
      chrome.debugger.sendCommand({ tabId: this.tabId }, method, params, (result) => {
        if (chrome.runtime.lastError) {
          reject(new Error(chrome.runtime.lastError.message));
        } else {
          resolve(result);
        }
      });
    });
  }
}

export function highlightTarget(el: HTMLElement): void {
  if (!el || typeof el.style === 'undefined') return;
  const originalOutline = el.style.outline;
  const originalOffset = el.style.outlineOffset;
  const originalBoxShadow = el.style.boxShadow;
  // Marvel's Vision Mind Stone gold + Android teal spotlight
  el.style.outline = '2px solid #fbbf24';
  el.style.outlineOffset = '2px';
  el.style.boxShadow = '0 0 14px rgba(251, 191, 36, 0.75)';

  }, 450);

/**
 * Dispatches a realistic PointerEvent + MouseEvent sequence with composed: true
 * for React Aria (Coursera), Material UI, and Shadow DOM (Cisco NetAcad).
 */
export function dispatchRealisticClick(el: HTMLElement): void {
  try {
    el.scrollIntoView({ block: 'center', inline: 'center', behavior: 'instant' });
  } catch {}

  const rect = el.getBoundingClientRect
    ? el.getBoundingClientRect()
    : { left: 10, top: 10, width: 20, height: 20 };
  const clientX = Math.round(rect.left + (rect.width || 20) / 2);
  const clientY = Math.round(rect.top + (rect.height || 20) / 2);

  const eventOpts: MouseEventInit = {
    bubbles: true,
    cancelable: true,
    composed: true,
    view: typeof window !== 'undefined' ? window : undefined,
    clientX,
    clientY,
    button: 0,
    buttons: 1,
  };

  if (typeof PointerEvent !== 'undefined') {
    el.dispatchEvent(new PointerEvent('pointerover', { ...eventOpts, pointerId: 1, isPrimary: true }));
    el.dispatchEvent(
      new PointerEvent('pointerenter', { ...eventOpts, bubbles: false, pointerId: 1, isPrimary: true })
    );
    el.dispatchEvent(new PointerEvent('pointerdown', { ...eventOpts, pointerId: 1, isPrimary: true }));
  }

  el.dispatchEvent(new MouseEvent('mouseover', eventOpts));
  el.dispatchEvent(new MouseEvent('mousedown', eventOpts));
  try {
    el.focus();
  } catch {}

  if (typeof PointerEvent !== 'undefined') {
    el.dispatchEvent(
      new PointerEvent('pointerup', { ...eventOpts, buttons: 0, pointerId: 1, isPrimary: true })
    );
  }

  el.dispatchEvent(new MouseEvent('mouseup', { ...eventOpts, buttons: 0 }));

  // If element is wrapped inside an <a> or <button>, ensure the interactive element receives the click
  const clickableTarget =
    el.tagName === 'A' || el.tagName === 'BUTTON' || el.tagName === 'INPUT'
      ? el
      : (el.closest?.('a, button, [role="button"], [role="link"], [role="checkbox"]') as HTMLElement | null) ||
        el;

  clickableTarget.click();
}

export function smartScroll(
  direction: 'up' | 'down',
  targetEl?: HTMLElement | null,
  amount = 450
): void {
  const deltaY = direction === 'down' ? amount : -amount;

  if (targetEl) {
    let cur: HTMLElement | null = targetEl;
    while (cur) {
      if (cur.scrollHeight > cur.clientHeight + 10) {
        cur.scrollTop += deltaY;
        try {
          cur.scrollBy({ top: deltaY, behavior: 'instant' });
        } catch {}
        return;
      }
      cur = cur.parentElement || ((cur.getRootNode?.() as ShadowRoot)?.host as HTMLElement) || null;
    }
  }

  if (typeof window !== 'undefined') {
    try {
      window.scrollBy({ top: deltaY, behavior: 'instant' });
    } catch {
      window.scrollBy(0, deltaY);
    }
  }

  if (typeof document !== 'undefined' && document.body) {
    const allNodes = collectAllElementsDeep(document.body, 2500);
    const scrollables: Array<{ el: HTMLElement; area: number }> = [];

    for (const node of allNodes) {
      if (node.scrollHeight <= node.clientHeight + 20) continue;
      if (typeof window !== 'undefined' && window.getComputedStyle) {
        const style = window.getComputedStyle(node);
        if (!['auto', 'scroll', 'overlay'].includes(style.overflowY)) continue;
      }
      const r = node.getBoundingClientRect ? node.getBoundingClientRect() : null;
      const area = r && r.width > 0 && r.height > 0 ? r.width * r.height : 1000;
      scrollables.push({ el: node, area });
    }

    scrollables.sort((a, b) => b.area - a.area);
    for (const item of scrollables.slice(0, 3)) {
      item.el.scrollTop += deltaY;
      try {
        item.el.scrollBy({ top: deltaY, behavior: 'instant' });
      } catch {}
    }
  }
}

export async function completePageMedia(targetEl?: HTMLElement | null): Promise<void> {
  if (typeof document === 'undefined' || !document.body) return;
  const allElements = collectAllElementsDeep(document.body, 2500);
  const mediaEls = allElements.filter(
    (el): el is HTMLMediaElement =>
      el.tagName.toUpperCase() === 'VIDEO' || el.tagName.toUpperCase() === 'AUDIO'
  );

  if (targetEl && (targetEl.tagName === 'VIDEO' || targetEl.tagName === 'AUDIO')) {
    if (!mediaEls.includes(targetEl as HTMLMediaElement)) {
      mediaEls.unshift(targetEl as HTMLMediaElement);
    }
  }

  for (const media of mediaEls) {
    try {
      media.muted = true;
      media.playbackRate = 2.0;
      const playPromise = media.play();
      if (playPromise && typeof playPromise.catch === 'function') {
        await playPromise.catch(() => {});
      }
      if (Number.isFinite(media.duration) && media.duration > 1) {
        media.currentTime = Math.max(0, media.duration - 0.25);
      }
      media.dispatchEvent(new Event('play', { bubbles: true }));
      media.dispatchEvent(new Event('playing', { bubbles: true }));
      media.dispatchEvent(new Event('timeupdate', { bubbles: true }));
      media.dispatchEvent(new Event('ended', { bubbles: true }));
    } catch {}
  }
}

export function getElementSelector(el: HTMLElement): string {
  if (el.id) {
    return `#${CSS?.escape ? CSS.escape(el.id) : el.id}`;
  }
  const role = el.getAttribute('role');
  if (role) {
    const name = el.getAttribute('aria-label') || el.getAttribute('name');
    if (name) {
      const esc = CSS?.escape ? CSS.escape(name) : name;
      return `[role="${role}"][aria-label="${esc}"]`;
    }
    return `[role="${role}"]`;
  }
  const name = el.getAttribute('name');
  if (name) {
    const esc = CSS?.escape ? CSS.escape(name) : name;
    return `${el.tagName.toLowerCase()}[name="${esc}"]`;
  }
  const ariaLabel = el.getAttribute('aria-label');
  if (ariaLabel) {
    const esc = CSS?.escape ? CSS.escape(ariaLabel) : ariaLabel;
    return `${el.tagName.toLowerCase()}[aria-label="${esc}"]`;
  }
  const testId = el.getAttribute('data-testid') || el.getAttribute('data-test');
  if (testId) {
    const esc = CSS?.escape ? CSS.escape(testId) : testId;
    return `[data-testid="${esc}"]`;
  }
  if (el.className && typeof el.className === 'string') {
    const classes = el.className
      .trim()
      .split(/\s+/)
      .filter((c) => !c.includes(':') && !c.includes('/'));
    if (classes.length > 0) {
      return `${el.tagName.toLowerCase()}.${classes
        .slice(0, 2)
        .map((c) => (CSS?.escape ? CSS.escape(c) : c))
        .join('.')}`;
    }
  }
  return el.tagName.toLowerCase();
}

export interface MacroExecutionResult {
  totalActions: number;
  executedActions: number;
  abortedEarly: boolean;
  abortReason?: AbortReason;
  executedTraces: ActionTrace[];
}

export interface MacroExecutorOptions {
  registry?: ElementRegistry;
  actuator?: CDPActuator;
  deltaTracker?: DOMDeltaTracker;
  debounceMs?: number;
  getCurrentUrl?: () => string;
}

export class MacroBatchExecutor {
  private registry: ElementRegistry;
  private actuator?: CDPActuator;
  private deltaTracker?: DOMDeltaTracker;
  private debounceMs: number;
  private getCurrentUrl: () => string;

  constructor(options?: MacroExecutorOptions) {
    this.registry = options?.registry ?? globalElementRegistry;
    this.actuator = options?.actuator;
    this.deltaTracker = options?.deltaTracker;
    this.debounceMs = options?.debounceMs ?? 140;
    this.getCurrentUrl =
      options?.getCurrentUrl ??
      (() => (typeof window !== 'undefined' ? window.location.href : ''));
  }

  async executeBatch(actions: MacroAction[]): Promise<MacroExecutionResult> {
    const executedTraces: ActionTrace[] = [];
    const initialUrl = this.getCurrentUrl();
    let lastTargetEl: HTMLElement | null = null;

    for (let i = 0; i < actions.length; i++) {
      const action = actions[i];

      if (action.stop_if_url_changes && this.getCurrentUrl() !== initialUrl) {
        return {
          totalActions: actions.length,
          executedActions: i,
          abortedEarly: true,
          abortReason: 'url_changed',
          executedTraces,
        };
      }

      if (this.deltaTracker) {
        const delta = this.deltaTracker.flushDelta();
        if (delta.hasAlertOrError) {
          return {
            totalActions: actions.length,
            executedActions: i,
            abortedEarly: true,
            abortReason: 'alert_or_error_mutated',
            executedTraces,
          };
        }
      }

      if (action.type === 'request_vision_crop') {
        executedTraces.push({
          action,
          timestamp: Date.now(),
          success: true,
        });
        return {
          totalActions: actions.length,
          executedActions: i + 1,
          abortedEarly: true,
          abortReason: 'vision_crop_requested',
          executedTraces,
        };
      }

      let targetEl: HTMLElement | null = null;
      if (action.id !== undefined) {
        targetEl = this.registry.getElement(action.id);
        if (
          (!targetEl || !targetEl.isConnected) &&
          action.type !== 'scroll' &&
          action.type !== 'complete_media'
        ) {
          return {
            totalActions: actions.length,
            executedActions: i,
            abortedEarly: true,
            abortReason: 'target_disappeared',
            executedTraces,
          };
        }
      } else if (action.type === 'press' && lastTargetEl && lastTargetEl.isConnected) {
        targetEl = lastTargetEl;
      }

      if (targetEl) {
        lastTargetEl = targetEl;
      }

      let selector: string | undefined;
      let elementText: string | undefined;
      if (targetEl) {
        selector = getElementSelector(targetEl);
        elementText = (targetEl.innerText || targetEl.textContent || '').trim().slice(0, 100);
      }

      try {
        await this.executeSingleAction(action, targetEl, lastTargetEl);
        executedTraces.push({
          action,
          timestamp: Date.now(),
          success: true,
          selector,
          elementText,
        });
      } catch (err) {
        executedTraces.push({
          action,
          timestamp: Date.now(),
          success: false,
          error: err instanceof Error ? err.message : String(err),
          selector,
          elementText,
        });

        return {
          totalActions: actions.length,
          executedActions: i + 1,
          abortedEarly: true,
          abortReason: 'action_failed',
          executedTraces,
        };
      }

      if (i < actions.length - 1 && this.debounceMs > 0) {
        await new Promise((r) => setTimeout(r, this.debounceMs));
      }
    }

    return {
      totalActions: actions.length,
      executedActions: actions.length,
      abortedEarly: false,
      executedTraces,
    };
  }

  private async executeSingleAction(
    action: MacroAction,
    targetEl: HTMLElement | null,
    lastTargetEl?: HTMLElement | null
  ): Promise<void> {
    const coords = action.id !== undefined ? this.registry.getCenterCoordinates(action.id) : null;

    switch (action.type) {
      case 'click': {
        if (targetEl) {
          highlightTarget(targetEl);
        }
        if (this.actuator) {
          const res = await this.actuator.click(coords, targetEl);
          if (!res.success) throw new Error(res.error ?? 'Click failed');
        } else if (targetEl) {
          dispatchRealisticClick(targetEl);
        } else {
          throw new Error(`Target element for click [${action.id}] not found`);
        }
        break;
      }

      case 'fill': {
        const val = action.value ?? '';
        if (targetEl) {
          highlightTarget(targetEl);
          try {
            targetEl.scrollIntoView({ block: 'center', inline: 'center', behavior: 'instant' });
          } catch {}
        }
        if (this.actuator) {
          const res = await this.actuator.type(val, coords, targetEl);
          if (!res.success) throw new Error(res.error ?? 'Fill failed');
        } else if (targetEl) {
          targetEl.focus();
          if (targetEl instanceof HTMLInputElement || targetEl instanceof HTMLTextAreaElement) {
            const proto =
              targetEl instanceof HTMLInputElement
                ? window.HTMLInputElement.prototype
                : window.HTMLTextAreaElement.prototype;
            const nativeSetter = Object.getOwnPropertyDescriptor(proto, 'value')?.set;
            if (nativeSetter) {
              nativeSetter.call(targetEl, val);
            } else {
              targetEl.value = val;
            }
            targetEl.dispatchEvent(new Event('input', { bubbles: true, composed: true }));
            targetEl.dispatchEvent(new Event('change', { bubbles: true, composed: true }));
          } else if (targetEl.isContentEditable) {
            targetEl.textContent = val;
            targetEl.dispatchEvent(new Event('input', { bubbles: true, composed: true }));
          }
        } else {
          throw new Error(`Target element for fill [${action.id}] not found`);
        }
        break;
      }

      case 'select': {
        if (targetEl) {
          highlightTarget(targetEl);
        }
        if (targetEl instanceof HTMLSelectElement) {
          targetEl.value = action.value ?? '';
          targetEl.dispatchEvent(new Event('change', { bubbles: true, composed: true }));
        } else if (targetEl) {
          dispatchRealisticClick(targetEl);
        } else {
          throw new Error(`Target select element [${action.id}] not found`);
        }
        break;
      }

      case 'press': {
        const key = action.value ?? 'Enter';
        const active =
          targetEl ||
          lastTargetEl ||
          (typeof document !== 'undefined'
            ? (document.activeElement as HTMLElement) || document.body
            : null);
        if (this.actuator) {
          const res = await this.actuator.pressKey(key, active);
          if (!res.success) throw new Error(res.error ?? 'Press failed');
        } else if (active) {
          const keyCode =
            key === 'Enter'
              ? 13
              : key === 'Tab'
              ? 9
              : key === 'Escape'
              ? 27
              : key === 'Backspace'
              ? 8
              : 0;
          const eventInit: KeyboardEventInit = {
            key,
            code: key,
            bubbles: true,
            cancelable: true,
            composed: true,
          };
          const keydown = new KeyboardEvent('keydown', eventInit);
          Object.defineProperty(keydown, 'keyCode', { get: () => keyCode });
          Object.defineProperty(keydown, 'which', { get: () => keyCode });
          active.dispatchEvent(keydown);

          const keypress = new KeyboardEvent('keypress', eventInit);
          Object.defineProperty(keypress, 'keyCode', { get: () => keyCode });
          Object.defineProperty(keypress, 'which', { get: () => keyCode });
          active.dispatchEvent(keypress);

          const keyup = new KeyboardEvent('keyup', eventInit);
          Object.defineProperty(keyup, 'keyCode', { get: () => keyCode });
          Object.defineProperty(keyup, 'which', { get: () => keyCode });
          active.dispatchEvent(keyup);

          if (key === 'Enter') {
            const form =
              active instanceof HTMLInputElement
                ? active.form
                : active.closest
                ? active.closest('form')
                : null;
            if (form) {
              try {
                if (typeof form.requestSubmit === 'function') {
                  form.requestSubmit();
                } else {
                  form.submit();
                }
              } catch {}
            }
          }
        }
        break;
      }

      case 'scroll': {
        const dir = action.value === 'up' ? 'up' : 'down';
        if (this.actuator && !targetEl) {
          await this.actuator.scroll(dir);
        }
        smartScroll(dir, targetEl);
        break;
      }

      case 'complete_media': {
        await completePageMedia(targetEl);
        break;
      }

      case 'wait': {
        const ms = action.value ? parseInt(action.value, 10) : 500;
        await new Promise((r) => setTimeout(r, isNaN(ms) ? 500 : ms));
        break;
      }

      default:
        break;
    }
  }
}

function fastHash(str: string): string {
  let hash = 5381;
  for (let i = 0; i < str.length; i++) {
    hash = (hash * 33) ^ str.charCodeAt(i);
  }
  return (hash >>> 0).toString(16);
}

export function estimateTokenCount(text: string): number {
  if (!text) return 0;
  return Math.ceil(text.length / 3.8);
}

export function estimateMessagesTokens(messages: ChatMessage[]): number {
  let total = 0;
  for (const msg of messages) {
    if (typeof msg.content === 'string') {
      total += estimateTokenCount(msg.content) + 4;
    } else {
      for (const block of msg.content) {
        if (block.type === 'text' && block.text) {
          total += estimateTokenCount(block.text);
        } else if (block.type === 'image') {
          total += 258;
        }
      }
      total += 4;
    }
  }
  return total;
}

export class ByteStableContextManager {
  private messages: ChatMessage[] = [];
  private initialPrefixHash = '';
  private prefixMessageCount = 0;
  private prefixTokenCount = 0;

  constructor() {
    this.messages = [];
  }

  initializeRun(goal: string, step1DistilledDom: string): void {
    this.messages = [
      {
        role: 'system',
        content: AGENT_SYSTEM_PROMPT,
        cache_control: { type: 'ephemeral' },
      },
      {
        role: 'user',
        content: `USER GOAL: ${goal}\n\nINITIAL VIEWPORT DOM (Step 1):\n${step1DistilledDom}`,
        cache_control: { type: 'ephemeral' },
      },
    ];

    this.prefixMessageCount = this.messages.length;
    this.initialPrefixHash = this.computePrefixHash();
    this.prefixTokenCount = estimateMessagesTokens(this.messages);
  }

  private computePrefixHash(): string {
    const prefixTexts = this.messages.slice(0, this.prefixMessageCount).map((m) => {
      return `${m.role}:${typeof m.content === 'string' ? m.content : JSON.stringify(m.content)}`;
    });
    return fastHash(prefixTexts.join('||'));
  }

  verifyPrefixStability(): boolean {
    if (this.messages.length < this.prefixMessageCount) {
      return false;
    }
    const currentPrefixHash = this.computePrefixHash();
    return currentPrefixHash === this.initialPrefixHash;
  }

  appendAssistantPlan(plan: AgentPlanResponse): void {
    if (!this.verifyPrefixStability()) {
      throw new Error('Prefix stability invariant violated before appending plan!');
    }

    const jsonContent = JSON.stringify(
      {
        thought: plan.thought,
        actions: plan.actions,
        complete: plan.complete,
      },
      null,
      2
    );

    this.messages.push({
      role: 'assistant',
      content: jsonContent,
    });
  }

  appendStepObservation(stepNumber: number, observationOrDelta: string): void {
    if (!this.verifyPrefixStability()) {
      throw new Error('Prefix stability invariant violated before appending observation!');
    }

    this.messages.push({
      role: 'user',
      content: `[STEP ${stepNumber} OBSERVATION]\n${observationOrDelta}`,
    });
  }

  getMessages(): ChatMessage[] {
    if (!this.verifyPrefixStability()) {
      throw new Error('Byte-stable prefix invariant violated!');
    }
    return [...this.messages];
  }

  getMessageCount(): number {
    return this.messages.length;
  }

  getPrefixTokenCount(): number {
    return this.prefixTokenCount;
  }

  getTelemetryMetrics(): TokenEstimationMetrics {
    const totalEstimatedTokens = estimateMessagesTokens(this.messages);
    const cachedPrefixTokens = this.prefixTokenCount;
    const freshTokens = Math.max(0, totalEstimatedTokens - cachedPrefixTokens);

    const billableEffective = freshTokens + cachedPrefixTokens * 0.1;
    const estimatedCostDiscountPercent =
      totalEstimatedTokens > 0
        ? Math.round(((totalEstimatedTokens - billableEffective) / totalEstimatedTokens) * 100)
        : 0;

    return {
      totalEstimatedTokens,
      cachedPrefixTokens,
      freshTokens,
      estimatedCostDiscountPercent,
    };
  }
}

import { AgentSessionState, ThinkingStep } from './types';
import { LLMRouter } from '../llm/router';
import { ByteStableContextManager } from '../llm/contextManager';
import { MacroAction } from '../llm/types';
import { ActionTrace, MacroExecutionResult, AbortReason } from '../executor/macroExecutor';

const DEFAULT_MAX_STEPS = 25;
const DEFAULT_STORAGE_SESSION_KEY = 'orbit_agent_session_state';
const SUBFRAME_ID_MULTIPLIER = 10000;

export interface StateMachineHooks {
  checkSkillCache?: (url: string, goal: string) => Promise<{ matched: boolean; actions?: MacroAction[] }>;
  isDestructiveAction?: (action: MacroAction, textDescription?: string) => { destructive: boolean; reason?: string };
  onStateChange?: (state: AgentSessionState) => void;
  sessionStorageKey?: string;
}

export class AgentStateMachine {
  private state: AgentSessionState;
  private router: LLMRouter;
  private contextManager: ByteStableContextManager;
  private hooks: StateMachineHooks;
  private storageKey: string;
  private lastKnownUrl = '';
  private actionHistorySignatures: string[] = [];
  private emptyActionCount = 0;
  private frameElementMap = new Map<number, { frameId: number; localId: number }>();

  constructor(router: LLMRouter, hooks: StateMachineHooks = {}, storageKey?: string) {
    this.router = router;
    this.hooks = hooks;
    this.storageKey = storageKey ?? hooks.sessionStorageKey ?? DEFAULT_STORAGE_SESSION_KEY;
    this.contextManager = new ByteStableContextManager();
    this.state = this.getInitialState();
  }

  private getInitialState(): AgentSessionState {
    return {
      status: 'idle',
      goal: '',
      tabId: 0,
      stepNumber: 0,
      maxSteps: DEFAULT_MAX_STEPS,
      currentThought: '',
      pendingActions: [],
      requiresApproval: false,
      tokenMetrics: {
        totalEstimatedTokens: 0,
        cachedPrefixTokens: 0,
        freshTokens: 0,
        estimatedCostDiscountPercent: 0,
      },
      completedTraces: [],
      thinkingHistory: [],
    };
  }

  getState(): Readonly<AgentSessionState> {
    return this.state;
  }

  async restoreSession(): Promise<boolean> {
    if (typeof chrome === 'undefined' || !chrome.storage?.session) return false;

    try {
      const res = await chrome.storage.session.get(this.storageKey);
      const saved = res[this.storageKey] as AgentSessionState | undefined;
      if (saved) {
        this.state = {
          ...saved,
          thinkingHistory: Array.isArray(saved.thinkingHistory) ? saved.thinkingHistory : [],
        };
        this.hooks.onStateChange?.(this.state);
        return true;
      }
    } catch (err) {
      console.warn('[AgentStateMachine] Failed to restore session from storage:', err);
    }
    return false;
  }

  private async persistSession(): Promise<void> {
    if (typeof chrome !== 'undefined' && chrome.storage?.session) {
      try {
        await chrome.storage.session.set({ [this.storageKey]: this.state });
      } catch (err) {
        console.warn('[AgentStateMachine] Failed to persist session:', err);
      }
    }
    this.hooks.onStateChange?.(this.state);
  }

  private upsertThinkingStep(step: ThinkingStep): void {
    const history = this.state.thinkingHistory ? [...this.state.thinkingHistory] : [];
    const idx = history.findIndex((s) => s.stepNumber === step.stepNumber);
    if (idx >= 0) {
      history[idx] = { ...history[idx], ...step };
    } else {
      history.push(step);
    }
    this.state.thinkingHistory = history;
  }

  async startTask(goal: string, tabId: number): Promise<void> {
    let targetTabId = tabId;

    if (targetTabId <= 0 && typeof chrome !== 'undefined' && chrome.tabs?.query) {
      try {
        const [activeTab] = await chrome.tabs.query({ active: true, lastFocusedWindow: true });
        if (activeTab?.id) {
          targetTabId = activeTab.id;
        } else {
          const allTabs = await chrome.tabs.query({ active: true });
          if (allTabs[0]?.id) targetTabId = allTabs[0].id;
        }
      } catch (err) {
        console.warn('[AgentStateMachine] Tab resolution fallback failed:', err);
      }
    }

    if (targetTabId <= 0 && typeof chrome !== 'undefined' && chrome.tabs) {
      this.state = {
        ...this.getInitialState(),
        status: 'error',
        goal,
        tabId: 0,
        error: 'No active browser tab found. Please make sure a web page is open and try again.',
      };
      await this.persistSession();
      return;
    }

    if (!this.hooks.sessionStorageKey && targetTabId > 0 && this.storageKey === DEFAULT_STORAGE_SESSION_KEY) {
      this.storageKey = `orbit_agent_session_tab_${targetTabId}`;
    }

    this.lastKnownUrl = '';
    this.actionHistorySignatures = [];
    this.emptyActionCount = 0;
    this.frameElementMap.clear();

    this.state = {
      ...this.getInitialState(),
      status: 'running',
      goal,
      tabId: targetTabId,
      stepNumber: 1,
      thinkingHistory: [],
    };
    await this.persistSession();

    await this.runStep();
  }

  async runStep(): Promise<AgentSessionState> {
    if (this.state.status !== 'running') {
      return this.state;
    }

    if (this.state.stepNumber > this.state.maxSteps) {
      this.state.status = 'error';
      this.state.error = `Maximum step limit (${this.state.maxSteps}) reached without task completion.`;
      await this.persistSession();
      return this.state;
    }

    const currentStepNum = this.state.stepNumber;

    try {
      // Emit live "thinking" step placeholder so Chat UI shows instant Mind Stone activity
      this.upsertThinkingStep({
        stepNumber: currentStepNum,
        thought: `Scanning page DOM and synthesizing Step ${currentStepNum} plan...`,
        actions: [],
        status: 'thinking',
        timestamp: Date.now(),
      });
      await this.persistSession();

      let observationText = '';
      if (currentStepNum === 1) {
        observationText = await this.queryTabDistillation(this.state.tabId);
        this.lastKnownUrl = await this.getTabUrl(this.state.tabId);

        if (this.hooks.checkSkillCache) {
          const tabUrl = this.lastKnownUrl;
          const skillMatch = await this.hooks.checkSkillCache(tabUrl, this.state.goal);
          if (skillMatch.matched && skillMatch.actions && skillMatch.actions.length > 0) {
            this.state.currentThought = 'Executing matched 0-token deterministic domain skill from cache.';
            const execResult = await this.sendActionsToTab(this.state.tabId, skillMatch.actions);
            this.state.completedTraces.push(...execResult.executedTraces);
            this.upsertThinkingStep({
              stepNumber: 1,
              thought: this.state.currentThought,
              actions: skillMatch.actions,
              executedTraces: execResult.executedTraces,
              status: 'done',
              timestamp: Date.now(),
              pageUrl: tabUrl,
            });
            this.state.status = 'completed';
            await this.persistSession();
            return this.state;
          }
        }

        this.contextManager.initializeRun(this.state.goal, observationText);
      } else {
        const previousUrl = this.lastKnownUrl;
        const currentUrl = await this.waitForTabSettled(this.state.tabId, previousUrl);
        const didNavigate = Boolean(currentUrl && previousUrl && currentUrl !== previousUrl);
        this.lastKnownUrl = currentUrl;

        // CRITICAL FIX: Always run queryTabDistillation so newly scrolled or expanded Coursera/LMS elements
        // get registered in globalElementRegistry with fresh [id] numbers!
        const deltaResult = await this.queryTabDelta(this.state.tabId);
        const freshDom = await this.queryTabDistillation(this.state.tabId);
        const hasNoDelta = deltaResult.includes('NO CHANGES');

        let stepObs = '';
        if (didNavigate) {
          stepObs = `ACTION RESULT: Page navigated to: ${currentUrl}\nCURRENT VIEWPORT DOM:\n${freshDom}`;
        } else if (hasNoDelta) {
          stepObs = `CURRENT PAGE: ${currentUrl || 'Current Tab'}\nCURRENT VIEWPORT DOM:\n${freshDom}`;
        } else {
          stepObs = `CURRENT PAGE: ${currentUrl || 'Current Tab'}\n${deltaResult}\nUPDATED VIEWPORT DOM:\n${freshDom}`;
        }

        if (this.actionHistorySignatures.length >= 2) {
          const lastSig = this.actionHistorySignatures[this.actionHistorySignatures.length - 1];
          const prevSig = this.actionHistorySignatures[this.actionHistorySignatures.length - 2];
          if (lastSig === prevSig && !didNavigate) {
            stepObs += `\n[SYSTEM NOTE: Previous action "${lastSig}" produced no page progression. Do NOT repeat it. Try scrolling down ({ "type": "scroll", "value": "down" }), completing media ({ "type": "complete_media" }), or clicking the next module/item.]`;
          }
        }

        observationText = stepObs;
        this.contextManager.appendStepObservation(currentStepNum, observationText);
      }

      this.updateTelemetry();

      const messages = this.contextManager.getMessages();
      let {
        response: plan,
        activeTier,
        provider,
        modelUsed,
        fallbackNotice,
      } = await this.router.executeStep(messages, 1);

      const isInfo = this.isPureInformationQuery(this.state.goal);
      const isRefusal = this.hasRefusalIndication(plan.thought);
      const isFinishedWithAnswer =
        (isInfo && this.hasCompletionIndication(plan.thought) && !isRefusal) ||
        (plan.complete === true && plan.thought.length > 20 && !isRefusal);

      if (currentStepNum === 1 && plan.actions.length === 0 && !isFinishedWithAnswer) {
        const actionNudge = `[SYSTEM ACTION MANDATE]: You returned 0 actions. As an autonomous browser agent, you must interact with the page. Do not output disclaimers or conversational lectures. Take the immediate next concrete step toward the goal (e.g. click the first module, lesson, start link, or accordion item visible in the viewport DOM). Output valid JSON with at least 1 action.`;
        this.contextManager.appendStepObservation(1, actionNudge);

        try {
          const nudgeResult = await this.router.executeStep(this.contextManager.getMessages(), 1);
          if (nudgeResult.response.actions.length > 0) {
            plan = nudgeResult.response;
            activeTier = nudgeResult.activeTier;
            provider = nudgeResult.provider;
            modelUsed = nudgeResult.modelUsed;
            if (nudgeResult.fallbackNotice) {
              fallbackNotice = nudgeResult.fallbackNotice;
            }
          }
        } catch (nudgeErr) {
          console.warn('[Vision AI] Action nudge execution failed:', nudgeErr);
        }
      }

      this.state.activeProvider = provider;
      this.state.activeTier = activeTier;
      this.state.activeModel = modelUsed;
      if (fallbackNotice) {
        this.state.fallbackNotice = fallbackNotice;
      }
      this.state.currentThought = plan.thought;
      this.state.pendingActions = plan.actions;

      this.upsertThinkingStep({
        stepNumber: currentStepNum,
        thought: plan.thought,
        actions: plan.actions,
        status: plan.actions.length > 0 ? 'acting' : 'done',
        timestamp: Date.now(),
        provider,
        model: modelUsed,
        pageUrl: this.lastKnownUrl,
      });
      await this.persistSession();

      if (this.hooks.isDestructiveAction) {
        for (const action of plan.actions) {
          const check = this.hooks.isDestructiveAction(action);
          if (check.destructive) {
            this.state.status = 'waiting_approval';
            this.state.requiresApproval = true;
            this.state.approvalDetails = {
              action,
              reason: check.reason ?? 'Potentially destructive or financial action requires user confirmation.',
            };
            await this.persistSession();
            return this.state;
          }
        }
      }

      await this.executePendingActions(plan.complete);
      return this.state;
    } catch (err) {
      const errMsg = err instanceof Error ? err.message : String(err);
      this.state.status = 'error';
      this.state.error = errMsg;
      this.upsertThinkingStep({
        stepNumber: currentStepNum,
        thought: this.state.currentThought || `Error on Step ${currentStepNum}: ${errMsg}`,
        actions: [],
        status: 'error',
        timestamp: Date.now(),
      });
      this.router.recordFailure();
      await this.persistSession();
      return this.state;
    }
  }

  async approvePendingAction(approved: boolean): Promise<void> {
    if (this.state.status !== 'waiting_approval') return;

    if (!approved) {
      this.state.status = 'paused';
      this.state.requiresApproval = false;
      this.state.approvalDetails = undefined;
      this.state.currentThought = 'User rejected destructive action. Execution paused.';
      await this.persistSession();
      return;
    }

    this.state.status = 'running';
    this.state.requiresApproval = false;
    this.state.approvalDetails = undefined;
    await this.persistSession();

    await this.executePendingActions(false);
  }

  private async executePendingActions(isPlanComplete: boolean): Promise<void> {
    const currentStepNum = this.state.stepNumber;
    const actions = this.state.pendingActions;
    this.state.pendingActions = [];

    const actionSig = actions.map((a) => `${a.type}:${a.id ?? ''}:${a.value ?? ''}`).join(';');
    if (actionSig) {
      this.actionHistorySignatures.push(actionSig);
    }

    if (actions.length === 0) {
      this.emptyActionCount += 1;
    } else {
      this.emptyActionCount = 0;
    }

    let stepTraces: ActionTrace[] = [];
    if (actions.length > 0) {
      const execResult = await this.sendActionsToTab(this.state.tabId, actions);
      if (execResult && Array.isArray(execResult.executedTraces)) {
        stepTraces = execResult.executedTraces;
        this.state.completedTraces.push(...execResult.executedTraces);
      }

      if (execResult.abortedEarly && execResult.abortReason !== 'vision_crop_requested') {
        this.router.recordFailure();
      } else {
        this.router.resetFailures();
      }
    }

    this.upsertThinkingStep({
      stepNumber: currentStepNum,
      thought: this.state.currentThought,
      actions,
      executedTraces: stepTraces,
      status: 'done',
      timestamp: Date.now(),
      provider: this.state.activeProvider,
      model: this.state.activeModel,
      pageUrl: this.lastKnownUrl,
    });

    let shouldComplete = isPlanComplete;

    if (this.hasRefusalIndication(this.state.currentThought)) {
      shouldComplete = false;
    }

    // Do not allow premature completion on multi-step goals if actions were just executed and thought indicates more steps remain
    if (
      shouldComplete &&
      actions.length > 0 &&
      /\b(now i will|next i will|opening|clicking module|let's open|scrolling to find)\b/i.test(
        this.state.currentThought
      )
    ) {
      shouldComplete = false;
    }

    if (
      actions.length === 0 &&
      this.emptyActionCount >= 2 &&
      this.hasRefusalIndication(this.state.currentThought)
    ) {
      this.state.status = 'error';
      this.state.error = this.state.currentThought;
      await this.persistSession();
      return;
    }

    const isActionGoal = !this.isPureInformationQuery(this.state.goal);
    const isInformationalAnswer =
      isPlanComplete &&
      this.state.currentThought.length > 20 &&
      !this.hasRefusalIndication(this.state.currentThought);
    if (this.state.stepNumber === 1 && actions.length === 0 && isActionGoal && !isInformationalAnswer) {
      shouldComplete = false;
    }

    if (!shouldComplete && actions.length === 0) {
      if (
        !this.hasRefusalIndication(this.state.currentThought) &&
        (this.hasCompletionIndication(this.state.currentThought) || this.emptyActionCount >= 2)
      ) {
        shouldComplete = true;
      }
    }

    if (!shouldComplete && this.actionHistorySignatures.length >= 3) {
      const len = this.actionHistorySignatures.length;
      if (
        this.actionHistorySignatures[len - 1] === this.actionHistorySignatures[len - 2] &&
        this.actionHistorySignatures[len - 2] === this.actionHistorySignatures[len - 3]
      ) {
        shouldComplete = true;
        this.state.currentThought = `${this.state.currentThought} (Autonomous loop breaker: task concluded as page reached a stable state).`;
      }
    }

    this.contextManager.appendAssistantPlan({
      thought: this.state.currentThought,
      actions,
      complete: shouldComplete,
    });

    if (shouldComplete) {
      this.state.status = 'completed';
      await this.persistSession();
      return;
    }

    this.state.stepNumber += 1;
    await this.persistSession();

    if (this.state.status === 'running') {
      await this.runStep();
    }
  }

  async pause(): Promise<void> {
    if (this.state.status === 'running' || this.state.status === 'waiting_approval') {
      this.state.status = 'paused';
      await this.persistSession();
    }
  }

  async resume(): Promise<void> {
    if (this.state.status === 'paused') {
      this.state.status = 'running';
      await this.persistSession();
      await this.runStep();

  private async appendSubframeDistillation(tabId: number, mainText: string): Promise<string> {
    if (typeof chrome === 'undefined' || !chrome.scripting?.executeScript || tabId <= 0) {
      return mainText;
    }

    try {
      const frameResults = await chrome.scripting.executeScript({
        target: { tabId, allFrames: true },
        func: () => {
          if (window === window.top) return null;
          if (window.innerWidth < 180 || window.innerHeight < 120) return null;

          const winAny = window as any;
          winAny.__vision_ai_subframe_reg__ =
            winAny.__vision_ai_subframe_reg__ || new Map<number, HTMLElement>();
          const reg: Map<number, HTMLElement> = winAny.__vision_ai_subframe_reg__;
          reg.clear();

          const walkDeep = (root: HTMLElement | ShadowRoot, out: HTMLElement[] = []): HTMLElement[] => {
            const ch = root.children;
            for (let i = 0; i < ch.length && out.length < 600; i++) {
              const el = ch[i] as HTMLElement;
              if (!el || !el.tagName) continue;
              out.push(el);
              if (el.shadowRoot) walkDeep(el.shadowRoot, out);
              if (el.children?.length) walkDeep(el, out);
            }
            return out;
          };

          const all = document.body ? walkDeep(document.body) : [];
          const items: Array<{ localId: number; desc: string }> = [];
          let localId = 1;

          for (const el of all) {
            if (items.length >= 25) break;
            const tag = el.tagName.toUpperCase();
            const role = (el.getAttribute('role') || '').toLowerCase();
            const isAction =
              ['A', 'BUTTON', 'INPUT', 'SELECT', 'TEXTAREA', 'VIDEO', 'SUMMARY'].includes(tag) ||
              ['button', 'link', 'checkbox', 'radio', 'tab', 'menuitem', 'option'].includes(role) ||
              el.hasAttribute('onclick') ||
              el.hasAttribute('aria-expanded');
            if (!isAction) continue;

            const r = el.getBoundingClientRect();
            if (r.width === 0 && r.height === 0) continue;

            const id = localId++;
            reg.set(id, el);
            const label = el.getAttribute('aria-label') || el.getAttribute('title') || '';
            const txt = (el.textContent || '').replace(/\s+/g, ' ').trim().slice(0, 32);
            const labelAttr = label ? ` label="${label.slice(0, 28)}"` : '';
            items.push({
              localId: id,
              desc: `<${tag.toLowerCase()}${labelAttr}>${txt}</${tag.toLowerCase()}>`,
            });
          }

          return items.length > 0 ? items : null;
        },
      });

      if (!Array.isArray(frameResults)) return mainText;

      const extraLines: string[] = [];
      let frameIndex = 0;
      for (const fr of frameResults) {
        const frameId = (fr as any).frameId as number | undefined;
        if (!frameId || frameId <= 0) continue;
        frameIndex++;

        const res = (fr as any).result as any;
        if (!res) continue;

        if (Array.isArray(res) && res.length > 0) {
          extraLines.push(`--- IFRAME [Frame ${frameId}: Embedded Content] ---`);
          for (const item of res) {
            const globalId = frameIndex * 1000 + item.localId;
            this.frameElementMap.set(globalId, { frameId, localId: item.localId });
            extraLines.push(`[${globalId}] ${item.desc}`);
          }
        } else if (res.items && Array.isArray(res.items) && res.items.length > 0) {
          const title = res.frameTitle || 'LMS Content';
          extraLines.push(`--- IFRAME [Frame ${frameId}: ${title}] ---`);
          for (const item of res.items) {
            const globalId = frameIndex * 1000 + item.localId;
            this.frameElementMap.set(globalId, { frameId, localId: item.localId });
            if (item.desc) {
              extraLines.push(`[${globalId}] ${item.desc}`);
            } else {
              const attrStr = item.attrStr
                ? item.attrStr.startsWith(' ')
                  ? item.attrStr
                  : ' ' + item.attrStr
                : '';
              if (item.text) {
                extraLines.push(`[${globalId}] <${item.tag}${attrStr}>${item.text}</${item.tag}>`);
              } else {
                extraLines.push(`[${globalId}] <${item.tag}${attrStr} />`);
              }
            }
          }
        }
      }

      if (extraLines.length > 0) {
        return `${mainText}\n${extraLines.join('\n')}`;
      }
    } catch {}

    return mainText;
  }

        target: { tabId, frameIds: [frameId] },
        func: (actionObj: MacroAction) => {

            winAny.__vision_ai_subframe_reg__ || winAny.__vision_ai_registry__;
          const el = actionObj.id !== undefined && reg ? reg.get(actionObj.id) : null;
          if (!el) return false;
          try {
            el.scrollIntoView({ block: 'center', inline: 'center', behavior: 'instant' });
          } catch {}
          if (actionObj.type === 'click') {
            el.dispatchEvent(new PointerEvent('pointerdown', { bubbles: true, composed: true }));
            el.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, composed: true }));
            el.dispatchEvent(new PointerEvent('pointerup', { bubbles: true, composed: true }));
            el.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, composed: true }));
            el.click();
          } else if (
            actionObj.type === 'fill' &&
            (el instanceof HTMLInputElement || el instanceof HTMLTextAreaElement)
          ) {
            el.value = actionObj.value || '';
            el.dispatchEvent(new Event('input', { bubbles: true, composed: true }));
            el.dispatchEvent(new Event('change', { bubbles: true, composed: true }));
          }
          return true;
        },
        args: [{ ...act, id: localId }],
      });
      return Boolean(res?.[0]?.result);
    } catch {
      return false;
    }
  }

  private async sendActionsToTab(tabId: number, actions: MacroAction[]): Promise<MacroExecutionResult> {

              success: false,
              error: String(backErr),

            abortReason = 'action_failed';

        } else {

              timestamp: Date.now(),

          try {

              (t) =>
                String(t.id) === q ||
                (t.url && t.url.toLowerCase().includes(q)) ||
                (t.title && t.title.toLowerCase().includes(q))

            executedTraces.push({
              action: act,
              timestamp: Date.now(),
              success: false,
              error: String(swErr),
            });
          }
        } else {
          executedTraces.push({ action: act, timestamp: Date.now(), success: true });
        }
      } else if (act.id !== undefined && this.frameElementMap.has(act.id)) {
        const mapped = this.frameElementMap.get(act.id)!;
        const ok = await this.executeSubframeAction(tabId, mapped.frameId, mapped.localId, act);
        executedTraces.push({ action: act, timestamp: Date.now(), success: ok });
      } else if (act.id !== undefined && act.id >= SUBFRAME_ID_MULTIPLIER) {
        const frameId = Math.floor(act.id / SUBFRAME_ID_MULTIPLIER);
        const localId = act.id % SUBFRAME_ID_MULTIPLIER;
        const ok = await this.executeSubframeAction(tabId, frameId, localId, act);
        executedTraces.push({ action: act, timestamp: Date.now(), success: ok });
      } else {
        domActions.push(act);
      }
    }

    if (domActions.length === 0 || abortedEarly) {
      return {
        totalActions: actions.length,
        executedActions: executedTraces.filter((t) => t.success).length,
        abortedEarly,
        abortReason,
        executedTraces,
      };
    }

    if (typeof chrome !== 'undefined' && tabId > 0) {

        try {
          const res = (await chrome.tabs.sendMessage(tabId, {
            type: 'EXECUTE_MACRO',
            actions: domActions,
          })) as Partial<MacroExecutionResult> | undefined;

          if (res) {
            const domTraces = Array.isArray(res.executedTraces)
              ? res.executedTraces
              : domActions.map((act) => ({ action: act, timestamp: Date.now(), success: true }));
            const combinedTraces = [...executedTraces, ...domTraces];
            return {
              totalActions: actions.length,
              executedActions: combinedTraces.filter((t) => t.success).length,
              abortedEarly: res.abortedEarly ?? false,
              abortReason: res.abortReason,
              executedTraces: combinedTraces,
            };
          }
        } catch (err) {
          console.warn(
            '[AgentStateMachine] Content script macro execution failed, attempting in-page execution:',
            err
          );
        }
      }
    }

    const combinedTraces = [
      ...executedTraces,
      ...domActions.map((act) => ({
        action: act,
        timestamp: Date.now(),
        success: true,
      })),
    ];
    return {
      totalActions: actions.length,
      executedActions: combinedTraces.filter((t) => t.success).length,
      abortedEarly: false,
      executedTraces: combinedTraces,
    };
  }

  const tabAgents = new Map<number, AgentStateMachine>();

  const getOrCreateTabAgent = (tabId: number): AgentStateMachine => {
    const key = tabId > 0 ? tabId : 0;
    const existing = tabAgents.get(key);
    if (existing) return existing;

    const machine = new AgentStateMachine(router, {
      sessionStorageKey: key > 0 ? `orbit_agent_session_tab_${key}` : 'orbit_agent_session_state',
      checkSkillCache: async (url, goal) => {
        const skill = await globalSkillStore.findMatchingSkill(url, goal);
        if (!skill) return { matched: false };
        await globalSkillStore.recordSkillExecution(skill.id);
        return {
          matched: true,
          actions: skill.steps.map((s) => ({
            type: s.action,
            value: s.value,
            stop_if_url_changes: s.stop_if_url_changes,
          })),
        };
      },
      isDestructiveAction: (action, text) => {
        const res = classifyActionSafety(action, text || '', '');
        return {
          destructive: res.isDestructive && res.severity !== 'low',
          reason: res.reason,
        };
      },
      onStateChange: (state) => {
        chrome.runtime
          .sendMessage({ type: 'AGENT_STATE_UPDATED', tabId: state.tabId || key, state })
          .catch(() => {});

        if (state.status === 'completed' && state.completedTraces.length >= 1) {
          const targetTab = state.tabId || key;
          const recordSkill = (tabUrl: string) => {
            compileTraceToSkill(tabUrl, state.goal, state.completedTraces, router).catch((err) => {
              console.warn('[Vision AI] Failed to compile trace to skill:', err);
            });
          };

          if (targetTab && typeof chrome !== 'undefined' && chrome.tabs?.get) {
            chrome.tabs
              .get(targetTab)
              .then((tab) => recordSkill(tab?.url || 'https://example.com'))
              .catch(() => recordSkill('https://example.com'));
          } else {
            recordSkill('https://example.com');
          }
        }
      },
    });

    tabAgents.set(key, machine);
    return machine;
  };

  if (typeof chrome !== 'undefined' && chrome.tabs?.onRemoved) {
    chrome.tabs.onRemoved.addListener((closedTabId) => {
      tabAgents.delete(closedTabId);
    });
  }

  const resolveActiveTabId = async (explicitTabId?: number): Promise<number> => {
    if (explicitTabId && explicitTabId > 0) return explicitTabId;
    try {
      const [lastFocused] = await chrome.tabs.query({ active: true, lastFocusedWindow: true });
      if (lastFocused?.id) return lastFocused.id;
    } catch {}
    try {
      const [activeTab] = await chrome.tabs.query({ active: true, currentWindow: true });
      if (activeTab?.id) return activeTab.id;
    } catch {}
    try {
      const allTabs = await chrome.tabs.query({ active: true });
      if (allTabs[0]?.id) return allTabs[0].id;
    } catch {}
    return 0;
  };

  chrome.runtime.onMessage.addListener((message: ExtensionMessage, _sender, sendResponse) => {
    if (!message || typeof message !== 'object') return false;

    switch (message.type) {
      case 'START_TASK': {
        resolveActiveTabId(message.tabId).then(async (tabId) => {
          await syncStorageToRouter();
          const agent = getOrCreateTabAgent(tabId);
          agent.startTask(message.goal, tabId).catch((err) => {
            console.warn(`[Vision AI] Tab ${tabId} startTask error:`, err);
          });
          sendResponse({ success: true, tabId });
        });
        return true;
      }

      case 'PAUSE_TASK': {
        resolveActiveTabId(message.tabId).then(async (tabId) => {
          const agent = getOrCreateTabAgent(tabId);
          await agent.pause();
          sendResponse({ success: true, tabId });
        });
        return true;
      }

      case 'RESUME_TASK': {
        resolveActiveTabId(message.tabId).then(async (tabId) => {
          await syncStorageToRouter();
          const agent = getOrCreateTabAgent(tabId);
          agent.resume().catch(() => {});
          sendResponse({ success: true, tabId });
        });
        return true;
      }

      case 'STOP_TASK': {
        resolveActiveTabId(message.tabId).then(async (tabId) => {
          const agent = getOrCreateTabAgent(tabId);
          await agent.stop();
          sendResponse({ success: true, tabId });
        });
        return true;
      }

      case 'GET_AGENT_STATE': {
        resolveActiveTabId(message.tabId).then(async (tabId) => {
          const agent = getOrCreateTabAgent(tabId);
          if (agent.getState().status === 'idle' && agent.getState().stepNumber === 0) {
            await agent.restoreSession();
          }
          sendResponse(agent.getState());
        });
        return true;
      }

      case 'SET_ACTIVE_MODEL': {
        (async () => {
          await syncStorageToRouter();
          const currentCfg = router.getConfig();
          const updates: Record<string, any> = {};
          if (message.provider === 'openrouter' || message.provider === 'groq' || message.provider === 'gemini') {
            updates.preferredTier1Provider = message.provider;
          }
          if (message.provider === 'openrouter') updates.openRouterModel = message.model;
          if (message.provider === 'groq') updates.groqModel = message.model;
          if (message.provider === 'gemini') updates.geminiModel = message.model;
          if (message.provider === 'anthropic') {
            updates.preferredTier2Provider = 'anthropic';
            updates.anthropicModel = message.model;
          }
          if (message.provider === 'ollama') updates.tier0Model = message.model;

          router.updateConfig(updates);
          if (typeof chrome !== 'undefined' && chrome.storage?.local) {
            const stored = await chrome.storage.local.get('modelRoutingConfig');
            await chrome.storage.local.set({
              modelRoutingConfig: { ...(stored.modelRoutingConfig || currentCfg), ...updates },
            });
          }
          sendResponse({ success: true, config: router.getConfig() });
        })();
        return true;
      }

      case 'INSPECT_TAB_DOM': {
        resolveActiveTabId(message.tabId).then(async (tabId) => {
          if (tabId <= 0 || typeof chrome === 'undefined' || !chrome.tabs?.sendMessage) {
            sendResponse({ error: 'No active tab to inspect' });
            return;
          }
          try {
            const res = await chrome.tabs.sendMessage(tabId, { type: 'DISTILL_VIEWPORT' });
            sendResponse({
              url: res?.url || '',
              title: res?.title || '',
              interactiveCount: res?.count || 0,
              distilledPreview: res?.text || '(Empty DOM)',
            });
          } catch (err) {
            sendResponse({ error: String(err) });
          }
        });
        return true;
      }

      case 'LIST_TAB_SESSIONS': {
        (async () => {
          const summaries: TabSessionSummary[] = [];
          try {
            const tabs = await chrome.tabs.query({ currentWindow: true });
            for (const t of tabs) {
              if (!t.id) continue;
              const url = t.url || '';
              const isInternal =
                url.startsWith('chrome://') ||
                url.startsWith('chrome-extension://') ||
                url.startsWith('edge://') ||
                url.startsWith('devtools://') ||
                url.startsWith('about:');
              const agent = tabAgents.get(t.id);
              const st = agent?.getState();
              summaries.push({
                tabId: t.id,
                title: t.title || (isInternal ? 'Chrome Internal' : 'Tab'),
                url,
                status: st?.status ?? 'idle',
                stepNumber: st?.stepNumber ?? 0,
                goal: st?.goal ?? '',
                isInternal,
              });
            }
          } catch {}
          sendResponse({ sessions: summaries });
        })();
        return true;
      }

      case 'APPROVE_ACTION': {
        resolveActiveTabId(message.tabId).then(async (tabId) => {
          const agent = getOrCreateTabAgent(tabId);
          await agent.approvePendingAction(message.approved);
          sendResponse({ success: true, tabId });
        });
        return true;
      }

      default:
        break;
    }

    return false;
  });
});

      }
    }

      const listener = (msg: { type: string; tabId?: number; state?: AgentSessionState }) => {
        if (msg && msg.type === 'AGENT_STATE_UPDATED' && msg.state) {
          const eventTabId = msg.tabId ?? msg.state.tabId;
          setState((prev) => {
            const targetId = pinnedTabId ?? activeTab.id ?? prev.tabId;
            if (!targetId || eventTabId === targetId) {
              return msg.state!;
            }
            return prev;
          });
          refreshTabSessions();
        }
      };

      chrome.runtime.onMessage.addListener(listener);

      const onTabActivated = () => refreshActiveTab();
      const onTabUpdated = () => refreshActiveTab();
      chrome.tabs?.onActivated?.addListener(onTabActivated);
      chrome.tabs?.onUpdated?.addListener(onTabUpdated);

      return () => {
        chrome.runtime.onMessage.removeListener(listener);
        chrome.tabs?.onActivated?.removeListener(onTabActivated);

    setSelectedModel(model);
    if (typeof chrome !== 'undefined' && chrome.runtime?.sendMessage) {

      });
    }
  };

  const handleStartTask = async (taskGoal?: string) => {
    const goalToRun = taskGoal ?? prompt;
    if (!goalToRun.trim()) return;

    setActiveSection('chat');
    let targetTabId = currentTabId;

    if (!pinnedTabId && typeof chrome !== 'undefined' && chrome.tabs?.query) {
      try {
        const [tab] = await chrome.tabs.query({ active: true, lastFocusedWindow: true });
        if (tab?.id) {
          targetTabId = tab.id;
          const url = tab.url || '';
          const isInternal =
            url.startsWith('chrome://') ||
            url.startsWith('chrome-extension://') ||
            url.startsWith('edge://') ||
            url.startsWith('devtools://') ||
            url.startsWith('about:');

              status: 'error',
              error:

          }
        }

      }

    }
  };

      chrome.runtime.sendMessage({ type: 'RESUME_TASK', tabId: currentTabId });
    }
  };

  };

  const handleReset = () => {
    if (typeof chrome !== 'undefined' && chrome.runtime?.sendMessage) {
      chrome.runtime.sendMessage({ type: 'STOP_TASK', tabId: currentTabId }, () => {
        setState({ ...INITIAL_STATE, tabId: currentTabId });
        setPrompt('');
        refreshTabSessions();
      });
    } else {
      setState(INITIAL_STATE);
      setPrompt('');
    }
  };

  const handleApprove = () => {
    if (typeof chrome !== 'undefined' && chrome.runtime?.sendMessage) {
      chrome.runtime.sendMessage({ type: 'APPROVE_ACTION', approved: true, tabId: currentTabId });
    }
  };

  const handleReject = () => {
    if (typeof chrome !== 'undefined' && chrome.runtime?.sendMessage) {
      chrome.runtime.sendMessage({ type: 'APPROVE_ACTION', approved: false, tabId: currentTabId });
    }
  };

  const handleRunSkill = (skill: SiteSkill) => {
    setPrompt(skill.goalPattern);
    handleStartTask(skill.goalPattern);
  };

  const runningBackgroundTabs = tabSessions.filter(
    (s) => s.status === 'running' || s.status === 'waiting_approval'
  );

  return (
    <div className="flex flex-col h-screen w-full bg-vision-bg text-slate-100 p-3 select-none font-sans overflow-hidden">
      {/* Marvel's Vision Synthezoid Header */}
      <header className="flex items-center justify-between border-b border-vision-border pb-2.5 mb-2">
        <div className="flex items-center gap-2.5">
          {/* Mind Stone Diamond Emblem */}
          <div className="relative flex items-center justify-center w-7 h-7 rounded-lg bg-gradient-to-br from-vision-crimson via-rose-800 to-vision-panel border border-vision-gold/60 mindstone-glow">
            <div className="w-2.5 h-2.5 rotate-45 bg-vision-gold shadow-sm" />
          </div>
          <div>
            <h1 className="text-sm font-black tracking-tight flex items-center gap-1.5 text-white">
              VISION AI
              <span className="text-[9px] bg-vision-crimson/20 text-vision-gold border border-vision-crimson/40 px-1.5 py-0.2 rounded font-mono font-semibold">
                SYNTHEZOID v2.0
              </span>
            </h1>
          </div>
        </div>

        <div className="flex items-center gap-1.5">
          <button
            onClick={() => setSkillsOpen(true)}
            className="flex items-center gap-1 text-xs bg-vision-panel hover:bg-vision-card border border-vision-border hover:border-vision-teal/50 px-2 py-1.5 rounded-lg text-slate-200 hover:text-vision-tealBright transition-colors"
            title="Saved 0-Token Domain Reflexes"
          >
            <Sparkles className="w-3.5 h-3.5 text-vision-tealBright" />
            <span className="text-[11px] font-semibold">Skills</span>
          </button>
          <button
            onClick={() => setVaultOpen(true)}
            className="flex items-center gap-1 text-xs bg-vision-panel hover:bg-vision-card border border-vision-border hover:border-vision-gold/50 px-2 py-1.5 rounded-lg text-slate-200 hover:text-vision-gold transition-colors"
            title="API Keys & Provider Vault"
          >
            <KeyRound className="w-3.5 h-3.5 text-vision-gold" />
            <span className="text-[11px] font-semibold">Vault</span>
          </button>
        </div>
      </header>

      {/* Multi-Tab Target Bar */}
      <div className="flex flex-col gap-1.5 bg-vision-panel/90 border border-vision-border rounded-xl px-2.5 py-1.5 mb-2 text-[11px]">
        <div className="flex items-center justify-between gap-2">
          <div className="flex items-center gap-1.5 min-w-0 flex-1">
            <Layers className="w-3.5 h-3.5 text-vision-crimsonLight shrink-0" />
            <span className="text-slate-400 shrink-0 font-semibold">Tab:</span>
            <select
              value={pinnedTabId ? String(pinnedTabId) : 'auto'}
              onChange={(e) => handleSelectTargetTab(e.target.value)}
              className="bg-vision-bg border border-vision-border rounded-lg px-2 py-0.5 text-[11px] text-slate-200 truncate flex-1 focus:border-vision-gold focus:outline-none"
            >
              <option value="auto">
                Active Tab ({activeTab.title?.slice(0, 28) || 'Current'})
              </option>
              {tabSessions
                .filter((t) => !t.isInternal)
                .map((t) => (
                  <option key={t.tabId} value={String(t.tabId)}>
                    {t.status === 'running' ? '⚡ [RUNNING] ' : ''}
                    {t.title.slice(0, 32)}
                  </option>
                ))}
            </select>
          </div>
          <button
            onClick={refreshActiveTab}
            title="Refresh open browser tabs"
            className="text-slate-400 hover:text-vision-gold p-1 rounded hover:bg-vision-card transition-colors shrink-0"
          >
            <RefreshCw className="w-3 h-3" />
          </button>
        </div>

        {runningBackgroundTabs.length > 0 && (
          <div className="flex items-center gap-1.5 overflow-x-auto pt-1 border-t border-vision-border/70">
            <span className="text-[10px] text-vision-gold font-semibold shrink-0">Running:</span>
            {runningBackgroundTabs.map((t) => (
              <button
                key={t.tabId}
                onClick={() => handleSelectTargetTab(String(t.tabId))}
                className={`text-[10px] px-2 py-0.5 rounded-full border flex items-center gap-1 shrink-0 transition-colors ${
                  currentTabId === t.tabId
                    ? 'bg-vision-crimson/30 text-white border-vision-gold'
                    : 'bg-vision-bg text-slate-300 border-vision-border hover:border-vision-gold/50'
                }`}
              >
                <span className="w-1.5 h-1.5 rounded-full bg-vision-gold animate-ping" />
                <span className="truncate max-w-[95px]">{t.title}</span>
                <span className="text-vision-gold font-mono">S{t.stepNumber}</span>
              </button>
            ))}
          </div>
        )}
      </div>

      {/* Segmented Section Switcher (Chat | Vision X-Ray | Telemetry | Vault) */}
      <nav className="grid grid-cols-4 gap-1 bg-vision-panel p-1 rounded-xl border border-vision-border mb-2.5 shrink-0">
        <button
          type="button"
          onClick={() => setActiveSection('chat')}
          className={`flex items-center justify-center gap-1 py-1.5 rounded-lg text-[11px] font-bold transition-all ${
            activeSection === 'chat'
              ? 'bg-gradient-to-r from-vision-crimson to-rose-700 text-white shadow-sm'
              : 'text-slate-400 hover:text-slate-200'
          }`}
        >
          <MessageSquare className="w-3 h-3" />
          <span>Chat</span>
        </button>

        <button
          type="button"
          onClick={() => setActiveSection('inspector')}
          className={`flex items-center justify-center gap-1 py-1.5 rounded-lg text-[11px] font-bold transition-all ${
            activeSection === 'inspector'
              ? 'bg-gradient-to-r from-vision-crimson to-rose-700 text-white shadow-sm'
              : 'text-slate-400 hover:text-slate-200'
          }`}
        >
          <Eye className="w-3 h-3" />
          <span>X-Ray</span>
        </button>

        <button
          type="button"
          onClick={() => setActiveSection('telemetry')}
          className={`flex items-center justify-center gap-1 py-1.5 rounded-lg text-[11px] font-bold transition-all ${
            activeSection === 'telemetry'
              ? 'bg-gradient-to-r from-vision-crimson to-rose-700 text-white shadow-sm'
              : 'text-slate-400 hover:text-slate-200'
          }`}
        >
          <Activity className="w-3 h-3" />
          <span>Telemetry</span>
          {state.tokenMetrics.estimatedCostDiscountPercent > 0 && (
            <span className="text-[9px] font-mono text-vision-gold">
              {state.tokenMetrics.estimatedCostDiscountPercent}%
            </span>
          )}
        </button>

        <button
          type="button"
          onClick={() => setActiveSection('skills')}
          className={`flex items-center justify-center gap-1 py-1.5 rounded-lg text-[11px] font-bold transition-all ${
            activeSection === 'skills'
              ? 'bg-gradient-to-r from-vision-crimson to-rose-700 text-white shadow-sm'
              : 'text-slate-400 hover:text-slate-200'
          }`}
        >
          <Shield className="w-3 h-3" />
          <span>System</span>
        </button>
      </nav>

      {activeTab.isInternal && !pinnedTabId && (
        <div className="flex items-start gap-1.5 bg-amber-950/40 border border-vision-gold/40 rounded-lg px-2.5 py-1.5 mb-2 text-[11px] text-amber-200">
          <AlertTriangle className="w-3.5 h-3.5 text-vision-gold shrink-0 mt-0.5" />
          <span>
            Switch to a regular website tab (e.g. Coursera, NetAcad, YouTube) or select a tab above.
          </span>
        </div>
      )}

      {/* SECTION 1: Main Chat + Model Selector + Thinking Steps */}
      {activeSection === 'chat' && (
        <TaskStream
          state={state}
          prompt={prompt}
          onPromptChange={setPrompt}
          onStart={(override) => handleStartTask(override)}
          onPause={handlePause}
          onResume={handleResume}
          onStop={handleStop}
          onReset={handleReset}
          selectedProvider={selectedProvider}
          selectedModel={selectedModel}
          onSelectModel={handleQuickSelectModel}
          onOpenVault={() => setVaultOpen(true)}
        />
      )}

      {/* SECTION 2: Live DOM X-Ray Inspector & 1-Click Coursera/LMS Macros */}
      {activeSection === 'inspector' && (
        <PageInspectorPanel
          tabId={currentTabId}
          onLaunchMacroGoal={(macroGoal) => {
            setPrompt(macroGoal);
            handleStartTask(macroGoal);
          }}
        />
      )}

      {/* SECTION 3: Dedicated Token Telemetry & Mind Stone Diagnostics */}
      {activeSection === 'telemetry' && (
        <div className="flex-1 overflow-y-auto">
          <TokenTelemetryHUD
            metrics={state.tokenMetrics}
            activeTier={state.activeTier}
            activeProvider={state.activeProvider || selectedProvider}
            activeModel={state.activeModel || selectedModel}
            stepCount={state.stepNumber}
            traceCount={state.completedTraces.length}
          />
        </div>
      )}

      {/* SECTION 4: System Hub (Quick access to 0-Token Skills & API Keys) */}
      {activeSection === 'skills' && (
        <div className="flex-1 flex flex-col gap-3 overflow-y-auto">
          <div className="bg-vision-panel border border-vision-border rounded-xl p-3.5 flex flex-col gap-2.5">
            <div className="text-xs font-bold text-vision-gold uppercase tracking-wider flex items-center gap-1.5">
              <KeyRound className="w-3.5 h-3.5" />
              <span>Multi-LLM API Vault & Failover</span>
            </div>
            <p className="text-[11px] text-slate-400">
              Configure API keys for Gemini, Groq, OpenRouter, Anthropic Claude, and Local Ollama with automatic 429/404 model failover.
            </p>
            <button
              type="button"
              onClick={() => setVaultOpen(true)}
              className="w-full py-2 rounded-lg bg-gradient-to-r from-vision-crimson to-rose-700 hover:from-rose-600 hover:to-rose-500 text-white font-bold text-xs shadow-md transition-all"
            >
              Open Full API Key & Model Vault
            </button>
          </div>

          <div className="bg-vision-panel border border-vision-border rounded-xl p-3.5 flex flex-col gap-2.5">
            <div className="text-xs font-bold text-vision-tealBright uppercase tracking-wider flex items-center gap-1.5">
              <Sparkles className="w-3.5 h-3.5" />
              <span>Zero-Token Learned Site Reflexes</span>
            </div>
            <p className="text-[11px] text-slate-400">
              Successful multi-step runs are automatically compiled into deterministic CSS/ARIA workflows in IndexedDB for 0-token replay.
            </p>
            <button
              type="button"
              onClick={() => setSkillsOpen(true)}
              className="w-full py-2 rounded-lg bg-vision-card hover:bg-vision-border text-vision-tealBright border border-vision-teal/40 font-bold text-xs transition-all"
            >
              Manage Saved 0-Token Skills
            </button>
          </div>
        </div>
      )}

      {/* Human-in-the-Loop Intercept Modal */}
      {state.requiresApproval && (
        <ApprovalModal
          action={state.approvalDetails?.action}
          reason={state.approvalDetails?.reason}
          onApprove={handleApprove}
          onReject={handleReject}
        />
      )}

      {/* Modals */}
      <ApiKeyVault isOpen={vaultOpen} onClose={() => setVaultOpen(false)} />
      <SkillsDrawer
        isOpen={skillsOpen}
        onClose={() => setSkillsOpen(false)}
        onRunSkill={handleRunSkill}
      />
    </div>
  );
}
